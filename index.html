function openTrainerMenuModal() { document.getElementById('trainerHoModal').style.display = 'flex'; populateModalTrainerDropdown(); populateModalMonthDropdown(); renderTrainerDashboard(); } function closeTrainerMenuModal() { document.getElementById('trainerHoModal').style.display = 'none'; } function populateModalTrainerDropdown() { const trainerSelect = document.getElementById('modalTrainerFilter'); trainerSelect.innerHTML = ''; let trainers = [...new Set(masterEmployees.map(item => item.trainerHo ? item.trainerHo.trim() : ""))].filter(Boolean); trainers.forEach(trainer => { let opt = document.createElement('option'); opt.value = trainer; opt.text = trainer; trainerSelect.appendChild(opt); }); } function populateModalMonthDropdown() { const monthSelect = document.getElementById('modalMonthFilter'); if (monthSelect.children.length > 0) return; // Salin opsi dari filter utama jika sudah ada const mainMonthSelect = document.getElementById('allStoresMonthFilter'); if (mainMonthSelect) { monthSelect.innerHTML = mainMonthSelect.innerHTML; monthSelect.value = mainMonthSelect.value; } } function renderTrainerDashboard() { const selectedTrainer = document.getElementById('modalTrainerFilter').value; const selectedMonth = document.getElementById('modalMonthFilter').value || new Date().toISOString().slice(0, 7); const searchVal = document.getElementById('modalStoreSearch').value.trim().toLowerCase(); const tableBody = document.getElementById('trainerModalTableBody'); let filteredMaster = masterEmployees; if (selectedTrainer !== "all") { filteredMaster = masterEmployees.filter(emp => emp.trainerHo && emp.trainerHo.trim().toLowerCase().includes(selectedTrainer.toLowerCase())); } let stores = [...new Set(filteredMaster.map(item => item.store ? item.store.trim() : ""))].filter(Boolean); if (searchVal) { stores = stores.filter(s => s.toLowerCase().includes(searchVal)); } if (stores.length === 0) { tableBody.innerHTML = `Tidak ada toko ditemukan untuk kriteria trainer ini.`; document.getElementById('trainerStatTotalStores').innerText = 0; document.getElementById('trainerStatCompleted').innerText = 0; document.getElementById('trainerStatAvgScore').innerText = "0 Poin"; return; } let totalStores = stores.length; let completedStoresCount = 0; let totalScoreSum = 0; let activeStoresWithScore = 0; let totalPhasesCount = Object.keys(phasesData).length; let rowsHTML = ""; stores.forEach(storeName => { let storeStaff = masterEmployees.filter(emp => emp.store && emp.store.trim().toLowerCase() === storeName.toLowerCase()); let trainerName = storeStaff.length > 0 && storeStaff[0].trainerHo ? storeStaff[0].trainerHo : "-"; let storeHistory = globalHistoryData.filter(item => { let cleanDate = item.coachingDate ? item.coachingDate.slice(0, 10) : ""; let itemMonth = cleanDate ? cleanDate.slice(0, 7) : ""; let jenisSesi = item.jenisSesi || "Coaching"; return item.store && item.store.trim().toLowerCase() === storeName.toLowerCase() && itemMonth === selectedMonth; }); let completedStaffCount = 0; let storeScoreAcc = 0; storeStaff.forEach(emp => { let empHist = storeHistory.filter(item => item.staffName && item.staffName.toLowerCase() === emp.name.trim().toLowerCase() && (item.jenisSesi || "Coaching") === "Coaching"); let empPhases = {}; empHist.forEach(h => { if (h.phaseName) empPhases[h.phaseName] = Number(h.totalScore || 0); }); let uniquePhases = Object.keys(empPhases).length; let empScore = Object.values(empPhases).reduce((a, b) => a + b, 0); storeScoreAcc += Math.min(empScore, 108); if (uniquePhases >= totalPhasesCount) completedStaffCount++; }); let avgScore = storeStaff.length > 0 && storeHistory.length > 0 ? Math.round(storeScoreAcc / storeStaff.length) : 0; if (storeHistory.length > 0) { totalScoreSum += avgScore; activeStoresWithScore++; if (storeStaff.length > 0 && completedStaffCount === storeStaff.length) completedStoresCount++; } let statusBadge = storeHistory.length === 0 ? 'Belum Submit' : (completedStaffCount === storeStaff.length ? 'Selesai 100%' : 'Dalam Proses'); rowsHTML += ` ${storeName} 👤 ${trainerName} ${storeStaff.length} Orang ${completedStaffCount} / ${storeStaff.length} ${avgScore} Poin ${statusBadge} `; }); let overallAvg = activeStoresWithScore > 0 ? Math.round(totalScoreSum / activeStoresWithScore) : 0; document.getElementById('trainerStatTotalStores').innerText = totalStores; document.getElementById('trainerStatCompleted').innerText = completedStoresCount; document.getElementById('trainerStatAvgScore').innerText = `${overallAvg} Poin`; document.getElementById('trainerModalTableBody').innerHTML = rowsHTML; }
